> with(numtheory):
> with(RandomTools):
> p:= nextprime(Generate(integer(range=3..10^3)));
> q:= nextprime(nextprime(p));
> 
> N:=p*q;
>
> k := 1+evalf(floor(N^(0.5)));
> y := (k^2)-N;
> d := 1;
> 
> halfN := N/2;
> while ((floor(y^(0.5)))<>(y^(0.5))) do
> y:= y+2*k+d:
> d:= d+2:
> if (floor(y^(0.5)))>=halfN then
> break;
> end if:
> end do:
> 
> x := (N+y)^(0.5);
> y := y^(0.5);
> 
> p1 := x-y;
